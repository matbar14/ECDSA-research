    93→Point add_safe(Point &a, Point &b) {
    94→  if (a.isZero()) return Point(b);
    95→  if (b.isZero()) return Point(a);
    96→  if (a.x.IsEqual(&b.x)) {
    97→    if (a.y.IsEqual(&b.y)) return curve.DoubleDirect(a);
    98→    Int y_sum(&a.y);
    99→    y_sum.Add(&b.y);
   100→    y_sum.Mod(&curve.P);
   101→    if (y_sum.IsZero()) {
   102→      Point infinity;
   103→      infinity.Clear();
   104→      return infinity;
   105→    }
   106→  }
   107→  return curve.AddDirect(a, b);
   108→}

   110→std::array<unsigned char, 33> point_key(Point &p) {
   111→  std::array<unsigned char, 33> raw{};
   112→  if (p.isZero()) {
   113→    return raw;
   114→  }
   115→  raw[0] = p.y.IsEven() ? 0x02 : 0x03;
   116→  alignas(8) unsigned char x_bytes[32];
   117→  p.x.Get32Bytes(x_bytes);
   118→  std::memcpy(raw.data() + 1, x_bytes, 32);
   119→  return raw;
   120→}

   122→std::array<unsigned char, 33> invariant_key(Point &p) {
   123→  std::array<unsigned char, 33> raw{};
   124→  if (p.isZero()) return raw;
   125→  Int cube;
   126→  cube.ModCube(&p.x);
   127→  alignas(8) unsigned char cube_bytes[32];
   128→  cube.Get32Bytes(cube_bytes);
   129→  std::memcpy(raw.data() + 1, cube_bytes, 32);
   130→  return raw;
   131→}

   137→Int lambda_value_for_x(Int x) {
   138→  if (x.IsZero()) {
   139→    Int zero;
   140→    zero.SetInt32(0);
   141→    return zero;
   142→  }
   143→  Int beta_x;
   144→  beta_x.ModMulK1(&beta, &x);
   145→  Int left_g;
   146→  left_g.ModSub(&curve.G.x, &beta_g);
   147→  Int left_x;
   148→  left_x.ModSub(&x, &beta_x);
   149→  Int numerator;
   150→  numerator.ModMulK1(&left_g, &left_x);
   151→
   152→  Int right_g;
   153→  right_g.ModSub(&curve.G.x, &beta_x);
   154→  Int right_x;
   155→  right_x.ModSub(&x, &beta_g);
   156→  Int denominator;
   157→  denominator.ModMulK1(&right_g, &right_x);
   158→  if (denominator.IsZero()) {
   159→    Int zero;
   160→    zero.SetInt32(0);
   161→    return zero;
   162→  }
   163→  denominator.ModInv();
   164→  numerator.ModMulK1(&denominator);
   165→  return numerator;
   166→}

   168→std::array<Int, 3> lambda_triangle_values(Point &point) {
   169→  Int x0;
   170→  if (!point.isZero()) x0.Set(&point.x);
   171→  else x0.SetInt32(0);
   172→  Int x1;
   173→  x1.ModMulK1(&beta, &x0);
   174→  Int x2;
   175→  x2.ModMulK1(&beta, &x1);
   176→  return {lambda_value_for_x(x0), lambda_value_for_x(x1),
   177→          lambda_value_for_x(x2)};
   178→}
   179→
   180→// Phase signature (d01, d12) of the ordered Lambda triple of p:
   181→//   d01 = Lambda(phi(p)) - Lambda(p), d12 = Lambda(phi^2(p)) - Lambda(phi(p)).
   182→// It is invariant under p -> -p and cyclically permuted by phi, so a bucket
   183→// match on K3=x^3 can be routed to one (+/- lambda^r) branch pair.
   184→std::array<unsigned char, 64> phase_signature_of(Point &p) {
   185→  std::array<unsigned char, 64> raw{};
   186→  if (p.isZero()) return raw;
   187→  std::array<Int, 3> values = lambda_triangle_values(p);
   188→  Int d01;
   189→  d01.ModSub(&values[1], &values[0]);
   190→  Int d12;
   191→  d12.ModSub(&values[2], &values[1]);
   192→  alignas(8) unsigned char tmp[32];
   193→  d01.Get32Bytes(tmp);
   194→  std::memcpy(raw.data(), tmp, 32);
   195→  d12.Get32Bytes(tmp);
   196→  std::memcpy(raw.data() + 32, tmp, 32);
   197→  return raw;
   198→}

   200→uint64_t ceil_sqrt_u64(uint64_t n) {
   201→  if (n <= 1) return 1;
   202→  uint64_t r = static_cast<uint64_t>(std::sqrt(static_cast<long double>(n)));
   203→  if (r == 0) r = 1;
   204→  while (r != UINT64_MAX && r < (n - 1) / r + 1) ++r;
   205→  while (r > 1 && (r - 1) >= (n - 1) / (r - 1) + 1) --r;
   206→  return r;
   207→}

   209→bool parse_u64(const char *s, uint64_t &out) {
   210→  if (!s || !*s) return false;
   211→  // Reject leading '+'/'-' and blanks: strtoull silently wraps "-1" into
   212→  // ULLONG_MAX, which used to be accepted as a valid width.
   213→  for (const char *p = s; *p; ++p) {
   214→    if (*p == '+' || *p == '-' || *p == ' ' || *p == '\t') return false;
   215→  }
   216→  char *end = nullptr;
   217→  unsigned long long value = std::strtoull(s, &end, 0);
   218→  if (*end != '\0') return false;
   219→  out = static_cast<uint64_t>(value);
   220→  return true;
   221→}

   223→bool parse_int(const std::string &text, Int &out) {
   224→  if (text.size() > 2 && text[0] == '0' &&
   225→      (text[1] == 'x' || text[1] == 'X')) {
   226→    out.SetBase16(text.c_str() + 2);
   227→    return true;
   228→  }
   229→  bool decimal = !text.empty();
   230→  for (char ch : text) decimal = decimal && ch >= '0' && ch <= '9';
   231→  if (!decimal) return false;
   232→  out.SetBase10(text.c_str());
   233→  return true;
   234→}

   236→bool parse_signed_int(const std::string &text, Int &out) {
   237→  if (text.empty()) return false;
   238→  if (text[0] != '-') return parse_int(text, out);
   239→  if (text.size() == 1) return false;
   240→  Int magnitude;
   241→  if (!parse_int(text.substr(1), magnitude)) return false;
   242→  out.Set(&curve.order);
   243→  out.Sub(&magnitude);
   244→  out.Mod(&curve.order);
   245→  return true;
   246→}

   254→void subtract_order_mod(Int &out, Int &value, Int &shift) {
   255→  out.Set(&value);
   256→  if (out.IsLower(&shift)) out.Add(&curve.order);
   257→  out.Sub(&shift);
   258→  out.Mod(&curve.order);
   259→}

   261→bool candidate_in_cover(const Int &candidate) {
   262→  if (!enforce_cover_range) return true;
   263→  Int value(candidate);

   267→bool check_orbit_candidate(Int &base_scalar, uint64_t j,
   268→                           Int &segment_end, Int &candidate_out,
   269→                           uint64_t &offset_out) {
   270→  Int factors[6];
   271→  factors[0].SetInt64(j);
   272→  factors[0].Mod(&curve.order);
   273→  set_scaled_mod(factors[1], glv_lambda, j);
   274→  set_scaled_mod(factors[2], glv_lambda2, j);
   275→  for (int i = 0; i < 3; ++i) {
   276→    factors[i + 3].Set(&curve.order);
   277→    factors[i + 3].Sub(&factors[i]);
   278→  }
   279→  for (Int &factor : factors) {
   280→    orbit_factor_checks.fetch_add(1, std::memory_order_relaxed);
   281→    Int candidate;
   282→    candidate.Set(&base_scalar);
   283→    candidate.Add(&factor);
   284→    candidate.Mod(&curve.order);
   285→    bool in_interval = !candidate.IsLower(&range_start) &&
   286→                       candidate.IsLower(&segment_end);
   287→    if (!in_interval && !partner_table_mode) continue;
   288→    exact_candidate_checks.fetch_add(1, std::memory_order_relaxed);
   289→    Point check = curve.ComputePublicKey(&candidate);
   290→    if (check.equals(search_target)) {
   291→      Int shifted_candidate(&candidate);
   292→      Int recovered;
   293→      subtract_order_mod(recovered, candidate, active_shift);
   294→      Point original_check = curve.ComputePublicKey(&recovered);
   295→      if (!original_check.equals(target) ||
   296→          (!partner_table_mode && !candidate_in_cover(recovered))) {
   297→        continue;
   298→      }
   299→      if (!in_interval) {
   300→        partner_recoveries.fetch_add(1, std::memory_order_relaxed);
   301→      }
   302→      candidate_out.Set(&recovered);
   303→      Int delta(&shifted_candidate);
   304→      delta.Sub(&range_start);
   305→      // Partner candidates sit far outside the 64-bit window offset range.
   306→      offset_out = in_interval ? delta.GetInt64() : 0;
   307→      return true;
   308→    }
   309→  }
   310→  return false;
   311→}

   313→// Route a K3 bucket match with the (d01, d12) phase signature: identify the
   314→// rotation r with giant = phi^r(+-row), then check only the two factors
   315→// (+-lambda^r * j). Falls back to the full six-branch check when the row
   316→// signature is degenerate, so coverage never depends on the router.
   317→bool check_orbit_candidate_signature(Int &base_scalar, uint64_t j,
   318→                                     size_t row_index,
   319→                                     std::array<unsigned char, 64> &giant_sig,
   320→                                     Int &segment_end, Int &candidate_out,
   321→                                     uint64_t &offset_out) {
   322→  const std::array<unsigned char, 64> &row_sig =
   323→      shared_signatures[row_index];
   324→  alignas(8) unsigned char r01_bytes[32];
   325→  alignas(8) unsigned char r12_bytes[32];
   326→  std::memcpy(r01_bytes, row_sig.data(), 32);
   327→  std::memcpy(r12_bytes, row_sig.data() + 32, 32);
   328→  Int r01, r12, rsum_neg;
   329→  r01.Set32Bytes(r01_bytes);
   330→  r12.Set32Bytes(r12_bytes);
   331→  // rsum_neg = -(r01 + r12) mod P, kept canonical in [0, P): Int::ModNeg
   332→  // maps 0 to P, which would break the byte comparison in the degenerate case.
   333→  rsum_neg.Set(&r01);
   334→  rsum_neg.Add(&r12);
   335→  rsum_neg.Mod(&curve.P);
   336→  if (!rsum_neg.IsZero()) {
   337→    rsum_neg.Neg();
   338→    rsum_neg.Add(&curve.P);
   339→  }
   340→  alignas(8) unsigned char g01_bytes[32];
   341→  alignas(8) unsigned char g12_bytes[32];
   342→  std::memcpy(g01_bytes, giant_sig.data(), 32);
   343→  std::memcpy(g12_bytes, giant_sig.data() + 32, 32);
   344→  Int g01, g12;
   345→  g01.Set32Bytes(g01_bytes);
   346→  g12.Set32Bytes(g12_bytes);
   347→
   348→  int route = -1;
   349→  if (g01.IsEqual(&r01) && g12.IsEqual(&r12)) {
   350→    route = 0;
   351→  } else if (g01.IsEqual(&r12) && g12.IsEqual(&rsum_neg)) {
   352→    route = 1;
   353→  } else if (g01.IsEqual(&rsum_neg) && g12.IsEqual(&r01)) {
   354→    route = 2;
   355→  }
   356→  if (route < 0) {
   357→    signature_anomalies.fetch_add(1, std::memory_order_relaxed);
   358→    return check_orbit_candidate(base_scalar, j, segment_end, candidate_out,
   359→                                 offset_out);
   360→  }
   361→  signature_routes.fetch_add(1, std::memory_order_relaxed);
   362→
   363→  Int base_factor;
   364→  if (route == 0) base_factor.SetInt64(j);
   365→  else if (route == 1) set_scaled_mod(base_factor, glv_lambda, j);
   366→  else set_scaled_mod(base_factor, glv_lambda2, j);
   367→  Int neg_factor;
   368→  neg_factor.Set(&curve.order);
   369→  neg_factor.Sub(&base_factor);
   370→
   371→  Int factors[2] = {base_factor, neg_factor};
   372→  for (Int &factor : factors) {
   373→    orbit_factor_checks.fetch_add(1, std::memory_order_relaxed);
   374→    Int candidate;
   375→    candidate.Set(&base_scalar);
   376→    candidate.Add(&factor);
   377→    candidate.Mod(&curve.order);
   378→    bool in_interval = !candidate.IsLower(&range_start) &&
   379→                       candidate.IsLower(&segment_end);
   380→    if (!in_interval && !partner_table_mode) continue;
   381→    exact_candidate_checks.fetch_add(1, std::memory_order_relaxed);
   382→    Point check = curve.ComputePublicKey(&candidate);
   383→    if (check.equals(search_target)) {
   384→      Int shifted_candidate(&candidate);
   385→      Int recovered;
   386→      subtract_order_mod(recovered, candidate, active_shift);
   387→      Point original_check = curve.ComputePublicKey(&recovered);
   388→      if (!original_check.equals(target) ||
   389→          (!partner_table_mode && !candidate_in_cover(recovered))) {
   390→        continue;
   391→      }
   392→      if (!in_interval) {
   393→        partner_recoveries.fetch_add(1, std::memory_order_relaxed);
   394→      }
   395→      candidate_out.Set(&recovered);
   396→      Int delta(&shifted_candidate);
   397→      delta.Sub(&range_start);
   398→      // Partner candidates sit far outside the 64-bit window offset range.
   399→      offset_out = in_interval ? delta.GetInt64() : 0;
   400→      return true;
   401→    }
   402→  }
   403→  return false;
   404→}

   406→void build_shared_table(uint64_t m) {
   407→  shared_table.resize(static_cast<size_t>(m));
   408→  const bool sig_mode = glv_orbit_mode && phase_signature;
   409→  if (sig_mode) shared_signatures.resize(static_cast<size_t>(m));
   410→  std::vector<std::thread> builders;
   411→  builders.reserve(worker_count);
   412→  for (unsigned worker = 0; worker < worker_count; ++worker) {
   413→    uint64_t begin = m * worker / worker_count;
   414→    uint64_t end = m * (worker + 1) / worker_count;
   415→    if (begin == end) continue;
   416→    builders.emplace_back([begin, end, sig_mode]() {
   417→      Point current;
   418→      if (begin == 0) {
   419→        current.Clear();
   420→      } else {
   421→        Int scalar;
   422→        scalar.SetInt64(begin);
   423→        current = curve.ComputePublicKey(&scalar);
   424→      }
   425→      for (uint64_t j = begin; j < end; ++j) {
   426→        shared_table[static_cast<size_t>(j)] = {
   427→            glv_orbit_mode ? invariant_key(current) : point_key(current), j};
   428→        if (sig_mode) {
   429→          shared_signatures[static_cast<size_t>(j)] = phase_signature_of(current);
   430→        }
   431→        current = add_safe(current, curve.G);
   432→      }
   433→    });
   434→  }
   435→  for (auto &builder : builders) builder.join();
   436→  if (sig_mode) {
   437→    // Sort rows and signatures together by key.
   438→    std::vector<uint64_t> order(static_cast<size_t>(m));
   439→    for (uint64_t i = 0; i < m; ++i) order[static_cast<size_t>(i)] = i;
   440→    std::sort(order.begin(), order.end(), [](uint64_t a, uint64_t b) {
   441→      return shared_table[static_cast<size_t>(a)].key <
   442→             shared_table[static_cast<size_t>(b)].key;
   443→    });
   444→    std::vector<BabyRow> sorted_rows;
   445→    std::vector<std::array<unsigned char, 64>> sorted_sigs;
   446→    sorted_rows.reserve(static_cast<size_t>(m));
   447→    sorted_sigs.reserve(static_cast<size_t>(m));
   448→    for (uint64_t idx : order) {
   449→      sorted_rows.push_back(shared_table[static_cast<size_t>(idx)]);
   450→      sorted_sigs.push_back(shared_signatures[static_cast<size_t>(idx)]);
   451→    }
   452→    shared_table.swap(sorted_rows);
   453→    shared_signatures.swap(sorted_sigs);
   454→  } else {
   455→    std::sort(shared_table.begin(), shared_table.end(), row_less);
   456→  }
   457→  uint64_t unique = m == 0 ? 0 : 1;
   458→  for (size_t i = 1; i < shared_table.size(); ++i) {
   459→    if (!(shared_table[i].key == shared_table[i - 1].key)) ++unique;
   460→  }
   461→  raw_baby_rows.store(m, std::memory_order_relaxed);
   462→  baby_rows.store(unique, std::memory_order_relaxed);
   463→}

   465→void search_slice(uint64_t local_start, uint64_t local_size,
   466→                  uint64_t segment) {
   467→  if (local_size == 0 || local_start >= active_range_size) return;
   468→  local_size = std::min(local_size, active_range_size - local_start);
   469→  uint64_t m = shared_baby_rows;
   470→  uint64_t giant_count = (local_size - 1) / m + 1;
   471→  giant_steps_planned.fetch_add(giant_count, std::memory_order_relaxed);
   472→
   473→  Int anchor_scalar(&range_start);
   474→  anchor_scalar.Add(local_start);
   475→  Int segment_end(&anchor_scalar);
   476→  segment_end.Add(local_size);
   477→  Point anchor = curve.ComputePublicKey(&anchor_scalar);
   478→  Point neg_anchor = curve.Negation(anchor);
   479→  Point giant_zero = add_safe(search_target, neg_anchor);
   480→
   481→  Int m_int(m);
   482→  Point step = curve.ComputePublicKey(&m_int);
   483→  Point neg_step = curve.Negation(step);
   484→
   485→  std::vector<std::thread> workers;
   486→  workers.reserve(worker_count);
   487→  for (unsigned worker = 0; worker < worker_count; ++worker) {
   488→    uint64_t begin = giant_count * worker / worker_count;
   489→    uint64_t end = giant_count * (worker + 1) / worker_count;
   490→    if (begin == end) continue;
   491→    workers.emplace_back([&, begin, end, segment, local_start]() {
   492→      Point giant = giant_zero;
   493→      Int base_scalar(&anchor_scalar);
   494→      if (begin != 0) {
   495→        Int delta;
   496→        delta.SetInt64(begin);
   497→        delta.Mult(m);
   498→        delta.Mod(&curve.order);
   499→        base_scalar.Add(&delta);
   500→        base_scalar.Mod(&curve.order);
   501→        Point shift = curve.ComputePublicKey(&delta);
   502→        Point neg_shift = curve.Negation(shift);
   503→        giant = add_safe(giant, neg_shift);
   504→      }
   505→      std::array<unsigned char, 64> giant_sig{};
   506→      bool giant_sig_ready = false;
   507→
   508→      for (uint64_t i = begin; i < end &&
   509→                              !found.load(std::memory_order_relaxed); ++i) {
   510→        giant_steps_scheduled.fetch_add(1, std::memory_order_relaxed);
   511→        giant_steps.fetch_add(1, std::memory_order_relaxed);
   512→        auto query_key = glv_orbit_mode ? invariant_key(giant) : point_key(giant);
   513→        auto it = std::lower_bound(shared_table.begin(), shared_table.end(), query_key,
   514→            [](const BabyRow &row, const std::array<unsigned char, 33> &needle) {
   515→              return row.key < needle;
   516→            });
   517→        bool bucket_checked = false;
   518→        while (it != shared_table.end() && it->key == query_key &&
   519→               !found.load(std::memory_order_relaxed)) {
   520→          if (glv_orbit_mode) {
   521→            invariant_hits.fetch_add(1, std::memory_order_relaxed);
   522→            if (bucket_checked) {
   523→              // Every row of a K3=x^3 bucket lies in one C6 orbit (the roots
   524→              // of X^3-c over F_p are exactly x, beta*x, beta^2*x), so the
   525→              // routed pair -- or the six GLV branches -- is the same scalar
   526→              // set for every row of the bucket. The first row's check
   527→              // already covered them; re-checking duplicates would repeat
   528→              // identical candidates. In non-GLV mode rows give distinct
   529→              // candidates (base + j), so no dedup is applied there.
   530→              orbit_duplicate_rows.fetch_add(1, std::memory_order_relaxed);
   531→              ++it;
   532→              continue;
   533→            }
   534→          }
   535→          uint64_t j = it->j;
   536→          Int candidate;
   537→          uint64_t local_offset = 0;
   538→          bool match = false;
   539→          if (glv_orbit_mode) {
   540→            if (phase_signature) {
   541→              if (!giant_sig_ready) {
   542→                giant_sig = phase_signature_of(giant);
   543→                giant_sig_ready = true;
   544→              }
   545→              match = check_orbit_candidate_signature(
   546→                  base_scalar, j, static_cast<size_t>(it - shared_table.begin()),
   547→                  giant_sig, segment_end, candidate, local_offset);
   548→            } else {
   549→              match = check_orbit_candidate(base_scalar, j, segment_end,
   550→                                            candidate, local_offset);
   551→            }
   552→          } else {
   553→            local_offset = local_start + i * m + j;
   554→            if (local_offset < local_start + local_size) {
   555→              candidate.Set(&base_scalar);
   556→              candidate.Add(j);
   557→              exact_candidate_checks.fetch_add(1, std::memory_order_relaxed);
   558→              Point check = curve.ComputePublicKey(&candidate);
   559→              match = check.equals(search_target);
   560→              if (match) {
   561→                Int recovered;
   562→                subtract_order_mod(recovered, candidate, active_shift);
   563→                Point original_check = curve.ComputePublicKey(&recovered);
   564→                match = original_check.equals(target) && candidate_in_cover(recovered);
   565→                if (match) {
   566→                  candidate.Set(&recovered);
   567→                }
   568→              }
   569→            }
   570→          }
   571→          if (match) {
   572→            std::lock_guard<std::mutex> lock(found_mutex);
   573→            if (!found.load()) {
   574→              found_scalar.Set(&candidate);
   575→              Int shifted_candidate(&candidate);
   576→              shifted_candidate.Add(&active_shift);
   577→              shifted_candidate.Mod(&curve.order);
   578→              found_shifted_scalar.Set(&shifted_candidate);
   579→              found_shift.Set(&active_shift);
   580→              found_segment = segment;
   581→              found_local_offset = local_offset;
   582→              found.store(true);
   583→            }
   584→            break;
   585→          }
   586→          if (glv_orbit_mode) {
   587→            bucket_checked = true;
   588→          }
   589→          ++it;
   590→        }
   591→        giant = add_safe(giant, neg_step);
   592→        base_scalar.Add(&m_int);
   593→        base_scalar.Mod(&curve.order);
   594→      }
   595→    });
   596→  }
   597→  for (auto &worker : workers) worker.join();
   598→  segments_done.fetch_add(1, std::memory_order_relaxed);
   599→}
   600→
   601→void search_segment(uint64_t segment) {
   602→  uint64_t local_start = segment * segment_size;
   603→  if (local_start >= active_range_size) return;
   604→  search_slice(local_start, std::min(segment_size, active_range_size - local_start),
   605→               segment);
   606→}
   607→
   608→std::vector<uint64_t> lambda_priority_order(Point &point) {
   609→  std::vector<uint64_t> result;
   610→  if (!lambda_priority || lambda_sector_count == 0) return result;
   611→  lambda_calculations.fetch_add(3, std::memory_order_relaxed);
   612→  std::array<Int, 3> values = lambda_triangle_values(point);
   613→  for (Int &value : values) {
   614→    value.Mult(lambda_sector_count);
   615→    value.Div(&curve.P);
   616→    uint64_t sector = value.GetInt64();
   617→    if (sector >= lambda_sector_count) sector = lambda_sector_count - 1;
   618→    if (std::find(result.begin(), result.end(), sector) == result.end()) {
   619→      result.push_back(sector);
   620→    }
   621→  }
   622→  return result;
   623→}
   624→
   625→void search_current_target() {
   626→  uint64_t sector_count = lambda_priority
   627→      ? std::min(lambda_sector_count, active_range_size == 0 ? uint64_t(1) : active_range_size)
   628→      : 0;
   629→  if (!lambda_priority) {
   630→    uint64_t segment_count = active_range_size / segment_size +
   631→                             (active_range_size % segment_size != 0);
   632→    for (uint64_t segment = 0; segment < segment_count &&
   633→                                 !found.load(std::memory_order_relaxed); ++segment) {
   634→      search_segment(segment);
   635→    }
   636→    return;
   637→  }
   638→
   639→  std::vector<uint64_t> priority = lambda_priority_order(search_target);
   640→  std::vector<bool> used(static_cast<size_t>(sector_count), false);
   641→  auto search_sector = [&](uint64_t sector, bool is_priority) {
   642→    if (sector >= sector_count || used[static_cast<size_t>(sector)]) return;
   643→    used[static_cast<size_t>(sector)] = true;
   644→    uint64_t begin = (active_range_size * sector) / sector_count;
   645→    uint64_t end = (active_range_size * (sector + 1)) / sector_count;
   646→    if (end <= begin) return;
   647→    if (is_priority) priority_slices.fetch_add(1, std::memory_order_relaxed);
   648→    else fallback_slices.fetch_add(1, std::memory_order_relaxed);
   649→    for (uint64_t pos = begin; pos < end &&
   650→                               !found.load(std::memory_order_relaxed); ) {
   651→      uint64_t amount = std::min(segment_size, end - pos);
   652→      search_slice(pos, amount, pos / segment_size);
   653→      pos += amount;
   654→    }
   655→  };
   656→  for (uint64_t sector : priority) {
   657→    if (found.load(std::memory_order_relaxed)) break;
   658→    search_sector(sector, true);
   659→  }
   660→  for (uint64_t sector = 0; sector < sector_count &&
   661→                             !found.load(std::memory_order_relaxed); ++sector) {
   662→    search_sector(sector, false);
   663→  }
   664→}

   716→    switch (c) {

   727→      case 'p': partner_table_mode = true; break;

   737→    usage(argv[0]); return 2;

   770→  if (!parse_int(start_text, range_start)) {
   771→    std::fprintf(stderr, "[E] start must be hexadecimal (0x...) or decimal\n");

   778→    return 2;

   792→      return 2;
   793→    }
   794→  }

   796→  mutable_key.push_back('\0');

   799→    std::fprintf(stderr, "[E] invalid SEC1 public key\n");
   800→    return 2;
   801→  }
   802→  if (lambda_dump) {
   803→    std::array<Int, 3> values = lambda_triangle_values(target);
   804→    for (size_t i = 0; i < values.size(); ++i) {
   805→      char *value = values[i].GetBase16();

   807→      std::free(value);
   808→    }
   809→    return 0;

   817→      std::fprintf(stderr, "[E] shift must be a signed hexadecimal or decimal scalar\n");
   818→      return 2;
   819→    }
   820→    requested_shifts.push_back(shift);
   821→  }
   822→

   824→  uint64_t m = ceil_sqrt_u64(std::min(segment_size, range_size));
   825→  shared_baby_rows = m;
   826→  std::cout << "Experimental bounded segmented BSGS\n"
   827→            << "  interval width: " << range_size << "\n"

   830→            << "  workers: " << worker_count << "\n"

   832→            << "  phase signature routing: " << (phase_signature ? "true" : "false") << "\n"
   833→            << "  partner table: " << (partner_table_mode ? "true" : "false") << "\n"
   834→            << "  shifted targets: " << requested_shifts.size()

   836→            << "  baby rows (shared): " << m << "\n"
   837→            << "  estimated shared table bytes: " << (m * sizeof(BabyRow)) << "\n"
   838→            << "  table copies: 1 (read-only, shared by workers)\n";
   839→  if (lambda_priority) {
   840→    std::cout << "  Lambda priority: true\n"
   841→              << "  Lambda sectors: " << lambda_sector_count << "\n";
   842→  }
   843→  if (automatic_coverage) {
   844→    char *cover_start_hex = cover_start.GetBase16();
   845→    char *cover_size_hex = cover_size.GetBase16();
   846→    std::cout << "  external cover start: 0x" << cover_start_hex << "\n"
   847→              << "  external cover size: 0x" << cover_size_hex << "\n";
   848→    std::free(cover_start_hex);
   849→    std::free(cover_size_hex);

   852→    std::cout << "  requested baby rows: " << requested_baby_rows << "\n";
